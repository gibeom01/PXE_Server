Jenkinsfile -> 이 파일을 Jenkins Pipeline Job 등록

pipeline {
    agent any

    // 물리 서버 작업 특성상 전체 타임아웃을 1시간으로 여유 있게 설정
    options {
        timeout(time: 1, unit: 'HOURS')
        ansiColor('xterm') // Ansible 로그를 컬러로 보기 위함
    }

    // Jenkins 웹 UI에서 작업자가 선택할 입력 값 정의
    parameters {
        string(name: 'IDRAC_IP', defaultValue: '192.168.0.120', description: '대상 서버의 iDRAC IP 주소')
        password(name: 'IDRAC_PASS', defaultValue: 'calvin', description: 'iDRAC 비밀번호')
        
        choice(name: 'RAID_LEVEL', choices: ['RAID1', 'RAID5', 'RAID10'], description: '구성할 RAID 레벨')
        choice(name: 'TARGET_OS', choices: ['rocky9', 'ubuntu22', 'windows2022'], description: '설치할 OS 종류')
    }

    stages {
        stage('1. 환경 검증') {
            steps {
                echo "배포 대상을 확인합니다: ${params.IDRAC_IP}"
                // iDRAC 포트와 네트워크 통신이 되는지 먼저 확인
                sh "ping -c 3 ${params.IDRAC_IP}"
            }
        }

        stage('2. iDRAC 작업 큐 초기화 (Ansible)') {
            steps {
                echo "이전 작업 찌꺼기를 제거하기 위해 iDRAC Job Queue를 비웁니다."
                // Ansible을 호출하여 큐 초기화 플레이북 실행
                sh """
                    ansible-playbook -i localhost, playbooks/clear_queue.yml \
                    -e "idrac_ip=${params.IDRAC_IP} idrac_password=${params.IDRAC_PASS}"
                """
            }
        }

        stage('3. 하드웨어 및 RAID 구성 (Ansible)') {
            steps {
                echo "RAID 설정을 시작합니다. 레벨: ${params.RAID_LEVEL}"
                // RAID 구성 플레이북 실행 (작업 완료 대기 루프가 이 내부에 포함됨)
                sh """
                    ansible-playbook -i localhost, playbooks/configure_raid.yml \
                    -e "idrac_ip=${params.IDRAC_IP} idrac_password=${params.IDRAC_PASS} raid_level=${params.RAID_LEVEL}"
                """
            }
        }

        stage('4. OS 무인 설치 이미지 마운트 및 부팅 (Ansible)') {
            steps {
                echo "선택하신 OS [${params.TARGET_OS}] 의 가상 미디어(ISO/스크립트)를 마운트합니다."
                // OS별 자동 설치 파일(ks.cfg 등)을 매핑하여 가상 미디어 부팅 트리거
                sh """
                    ansible-playbook -i localhost, playbooks/deploy_os.yml \
                    -e "idrac_ip=${params.IDRAC_IP} idrac_password=${params.IDRAC_PASS} target_os=${params.TARGET_OS}"
                """
            }
        }
        
        stage('5. OS 설치 모니터링') {
            steps {
                echo "물리 서버가 재부팅되어 OS 자동 설치를 진행 중입니다."
                echo "이 단계는 가상 미디어가 정상 물린 후 서버가 켜지면 무인으로 진행되므로,"
                echo "작업자는 물리 서버 전면 모니터나 iDRAC 가상 콘솔화면을 통해 설치 완료를 확인하세요."
            }
        }
    }

    post {
        success {
            echo "👍 [성공] iDRAC 제어 및 OS 배포 명령이 성공적으로 완료되었습니다."
        }
        failure {
            echo "❌ [실패] 배포 중 에러가 발생했습니다. Jenkins 콘솔 로그를 확인하세요."
        }
    }
}
