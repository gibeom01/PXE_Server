playbooks/configure_raid.yml -> Jenkins 호출하는 실제 Ansible 플레이북 핵심 예

---
- name: Dell R630 RAID Configuration via iDRAC
  hosts: localhost
  connection: local
  gather_facts: no

  tasks:
    # 1. RAID 설정 명령 투척 (명령을 내리면 iDRAC이 Job ID를 리턴함)
    - name: Create Virtual Disk (RAID Configuration)
      dellemc.openmanage.idrac_storage_volume:
        idrac_ip: "{{ idrac_ip }}"
        idrac_user: "root"
        idrac_password: "{{ idrac_password }}"
        share_mnt: "/var/lib/tftpboot" # 노트북 내 로기 가상 경로 설정 가능
        state: "present"
        volumes:
          - RAIDType: "{{ raid_level }}"
            VolumeName: "OS_VD"
      register: raid_job_result

    # 2. [가장 중요] 리턴받은 Job ID를 가지고 완료될 때까지 끈질기게 대기
    - name: Wait for RAID Job to complete (Timeout: 15 Mins)
      dellemc.openmanage.idrac_job_queue:
        idrac_ip: "{{ idrac_ip }}"
        idrac_user: "root"
        idrac_password: "{{ idrac_password }}"
        job_id: "{{ raid_job_result.job_id }}"
        wait: yes                  # 완료될 때까지 태스크를 종료하지 않고 대기하겠다는 옵션
        wait_timeout: 900          # 최대 15분(900초) 동안 iDRAC 상태를 조회하며 대기
